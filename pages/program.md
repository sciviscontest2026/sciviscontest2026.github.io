// On the event day, mark the session running now and the one up next (Boston time).
// Preview with ?now=2026-11-09T09:05
(function () {
    var EVENT_DAY = '2026-11-09';
    var items = document.querySelectorAll('#program-agenda .program-item');
    if (!items.length) return;

    function toMinutes(hm) {
        var p = hm.split(':');
        return parseInt(p[0], 10) * 60 + parseInt(p[1], 10);
    }

    function bostonNow() {
        var m = /[?&]now=(\d{4}-\d{2}-\d{2})T(\d{1,2}:\d{2})/.exec(location.search);
        if (m) return { day: m[1], minutes: toMinutes(m[2]) };
        var parts = {};
        new Intl.DateTimeFormat('en-US', {
            timeZone: 'America/New_York', hourCycle: 'h23',
            year: 'numeric', month: '2-digit', day: '2-digit', hour: '2-digit', minute: '2-digit'
        }).formatToParts(new Date()).forEach(function (p) { parts[p.type] = p.value; });
        return {
            day: parts.year + '-' + parts.month + '-' + parts.day,
            minutes: parseInt(parts.hour, 10) * 60 + parseInt(parts.minute, 10)
        };
    }

    function tag(item, cls, text) {
        var el = document.createElement('span');
        el.className = 'program-live ' + cls;
        el.textContent = text;
        var title = item.querySelector('.program-title');
        title.insertBefore(el, title.querySelector('.program-who'));
    }

    function update() {
        Array.prototype.forEach.call(items, function (item) {
            item.classList.remove('is-now', 'is-next');
            var old = item.querySelector('.program-live');
            if (old) old.remove();
        });

        var now = bostonNow();
        if (now.day !== EVENT_DAY) return;

        for (var i = 0; i < items.length; i++) {
            var start = toMinutes(items[i].dataset.start);
            var end = toMinutes(items[i].dataset.end);
            if (now.minutes >= start && now.minutes < end) {
                items[i].classList.add('is-now');
                tag(items[i], 'live-now', 'Now');
                if (items[i + 1]) {
                    items[i + 1].classList.add('is-next');
                    tag(items[i + 1], 'live-next', 'Up next');
                }
                return;
            }
            if (now.minutes < start) {
                items[i].classList.add('is-next');
                tag(items[i], 'live-next', 'Up next');
                return;
            }
        }
    }

    update();
    setInterval(update, 30000);
})();
</script>
